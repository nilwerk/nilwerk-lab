# Host unit tests for the sensor drivers (task 4.F1). No ESP-IDF, no hardware: gcc/clang + cmake + python3.
# CI: cmake -S test/host -B build-host && cmake --build build-host && ctest --test-dir build-host --output-on-failure
cmake_minimum_required(VERSION 3.16)
project(nw_sensors_host C)
set(CMAKE_C_STANDARD 11)
set(SENS ${CMAKE_CURRENT_SOURCE_DIR}/../../components/sensors)
set(DRV ${SENS}/src/nw_aht20.c ${SENS}/src/nw_bh1750.c ${SENS}/src/nw_soil.c ${SENS}/src/nw_battery.c)
add_executable(test_sensors test_main.c ${DRV})
target_include_directories(test_sensors PRIVATE ${SENS}/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(test_sensors PRIVATE -Wall -Wextra -Werror)
if(NOT MSVC)
  target_compile_options(test_sensors PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  target_link_options(test_sensors PRIVATE -fsanitize=address,undefined)
endif()
# Provisioning logic (task 4.F2)
set(PROV ${CMAKE_CURRENT_SOURCE_DIR}/../../components/provisioning)
add_executable(test_prov test_prov.c ${PROV}/src/nw_prov_proto.c ${PROV}/src/nw_prov.c)
target_include_directories(test_prov PRIVATE ${PROV}/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(test_prov PRIVATE -Wall -Wextra -Werror)
if(NOT MSVC)
  target_compile_options(test_prov PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  target_link_options(test_prov PRIVATE -fsanitize=address,undefined)
endif()
# Glue core (task 4.F2 slice 2)
add_executable(test_glue test_glue.c ${PROV}/src/nw_prov_proto.c ${PROV}/src/nw_prov.c ${PROV}/src/nw_prov_glue.c)
target_include_directories(test_glue PRIVATE ${PROV}/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(test_glue PRIVATE -Wall -Wextra -Werror)
if(NOT MSVC)
  target_compile_options(test_glue PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  target_link_options(test_glue PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME glue_unit COMMAND test_glue)
add_test(NAME sensors_unit COMMAND test_sensors)
add_test(NAME provisioning_unit COMMAND test_prov)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME board_pins_cite_pinmap
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_board_pins.py)
  add_test(NAME provisioning_contract
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_prov_contract.py)
  add_test(NAME provisioning_mutation_checks
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/mutate_prov.py ${CMAKE_C_COMPILER})
  add_test(NAME mutation_checks
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/mutate.py ${CMAKE_C_COMPILER})
endif()
